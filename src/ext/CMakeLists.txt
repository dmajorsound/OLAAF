# olAAF REAPER extension: reaper_olaaf.dll / .dylib / .so

set(EXT_SRC main.cpp progress.cpp dialog.cpp theme.cpp logo.cpp fonts.cpp round.cpp update.cpp)
if(WIN32 OR APPLE)
  list(APPEND EXT_SRC fonts_data.cpp) # IBM Plex, embedded
endif()
set(SWELL ${PROJECT_SOURCE_DIR}/third_party/reaper-sdk/WDL/swell)
if(APPLE)
  list(APPEND EXT_SRC ${SWELL}/swell-modstub.mm)
  set_source_files_properties(${SWELL}/swell-modstub.mm PROPERTIES COMPILE_FLAGS "-x objective-c++")
elseif(UNIX)
  list(APPEND EXT_SRC ${SWELL}/swell-modstub-generic.cpp)
endif()
add_library(reaper_olaaf MODULE ${EXT_SRC})
target_link_libraries(reaper_olaaf PRIVATE olaaf_core)
# reaper_plugin.h finds "../WDL/swell/swell.h" next to it on mac/linux
target_include_directories(reaper_olaaf PRIVATE ${PROJECT_SOURCE_DIR}/third_party/reaper-sdk/sdk)
set_target_properties(reaper_olaaf PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)

if(NOT WIN32)
  target_compile_definitions(reaper_olaaf PRIVATE SWELL_PROVIDED_BY_APP)
else()
  target_link_libraries(reaper_olaaf PRIVATE shell32 ole32 gdi32 gdiplus winhttp)
endif()

if(APPLE)
  set_target_properties(reaper_olaaf PROPERTIES SUFFIX ".dylib")
  target_link_options(reaper_olaaf PRIVATE -undefined dynamic_lookup)
  target_link_libraries(reaper_olaaf PRIVATE "-framework AppKit" "-framework CoreText" "-framework CoreGraphics")
elseif(UNIX)
  set_target_properties(reaper_olaaf PROPERTIES SUFFIX ".so")
  target_link_libraries(reaper_olaaf PRIVATE pthread dl)
endif()

# update check over HTTPS: WinHTTP on Windows, the system libcurl on macOS (and Linux, if present)
if(APPLE)
  target_link_libraries(reaper_olaaf PRIVATE curl)
  target_compile_definitions(reaper_olaaf PRIVATE OLAAF_HAVE_CURL)
elseif(UNIX)
  find_package(CURL QUIET)
  if(CURL_FOUND)
    target_link_libraries(reaper_olaaf PRIVATE CURL::libcurl)
    target_compile_definitions(reaper_olaaf PRIVATE OLAAF_HAVE_CURL)
  endif()
endif()
